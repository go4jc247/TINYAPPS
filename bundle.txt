// Hub service worker (landing page). One shared cache name for the whole repo, so the hub can
// refresh every app's files and each app's own worker reads the same up-to-date copies.
const CACHE = 'tiny-apps';
const FILES = ['./', './index.html', './manifest.webmanifest', './icon.png'];

self.addEventListener('install', e => {
  e.waitUntil(caches.open(CACHE).then(c => c.addAll(FILES)).then(() => self.skipWaiting()));
});

self.addEventListener('activate', e => {
  e.waitUntil(self.clients.claim());
});

// Network first (updates arrive), cache when offline.
self.addEventListener('fetch', e => {
  const req = e.request;
  if (req.method !== 'GET' || new URL(req.url).origin !== location.origin) return;
  e.respondWith(
    fetch(req)
      .then(res => {
        const copy = res.clone();
        caches.open(CACHE).then(c => c.put(req, copy));
        return res;
      })
      .catch(() => caches.match(req, { ignoreSearch: true }).then(r => r || caches.match('./index.html')))
  );
});

function mergedState(snaps, upto) {
  const m = { fields: {}, ts: {}, payments: [], paymentsTs: 0, autopay: '' };
  for (let i = 0; i <= upto && i < snaps.length; i++) {
    const s = snaps[i];
    for (const k in s.fields) {
      if (isBlank(s.fields[k]) && m.fields[k] !== undefined) continue;
      m.fields[k] = s.fields[k]; m.ts[k] = s.ts;
    }
    if (s.payments && s.payments.length) { m.payments = s.payments; m.paymentsTs = s.ts; }
    if (s.autopay) m.autopay = s.autopay;
  }
  return m;
}
function parseDue(v) {
  const m = String(v || '').match(/^(\d{1,2})\/(\d{1,2})\/(\d{2,4})$/);
  if (!m) return null;
  return new Date(m[3].length === 4 ? +m[3] : 2000 + +m[3], +m[1] - 1, +m[2]).getTime();
}
// Days relative to the due date as of "now". Negative = days until due, positive = days past due.
// Uses whichever of "Days Past Due" / "Current Due Date" was pasted most recently (a tie goes to
// the lender's own Days Past Due number, moved forward by the days since that paste).
function daysNow(m, now) {
  const pd = parseInt(m.fields['Days Past Due'], 10), pts = m.ts['Days Past Due'];
  const due = parseDue(m.fields['Current Due Date']), dts = m.ts['Current Due Date'];
  const hasPd = !isNaN(pd) && pts;
  if (due !== null && (!hasPd || dts > pts)) return { n: Math.round((dayStart(now) - due) / DAY), src: 'due' };
  if (hasPd) return { n: pd + Math.round((dayStart(now) - dayStart(pts)) / DAY), src: 'paste' };
  return null;
}
function payStr(p) { return [p.date, p.type, p.account, p.amount].filter(Boolean).join(' · '); }
function recStr(r) { return [r.date, r.status, r.note, r.amount, r.account].filter(Boolean).join(' · '); }
function diffSnaps(a, b) {
  const out = [];
  const keys = [...new Set([...Object.keys(a ? a.fields : {}), ...Object.keys(b.fields)])];
  for (const k of keys) {
    const o = a ? a.fields[k] : undefined, n = b.fields[k];
    if (o !== n) {
      // the days count moves on its own each day, so don't report it as a change
      if (k === 'Days Past Due' && a && a.ts[k] && b.ts[k] && parseInt(o, 10) + Math.round((dayStart(b.ts[k]) - dayStart(a.ts[k])) / DAY) === parseInt(n, 10)) continue;
      out.push({ k, o, n });
    }
  }
  if ((a ? a.autopay : '') !== b.autopay) out.push({ k: 'Autopay', o: a ? a.autopay : undefined, n: b.autopay });
  const po = a ? a.payments.map(payStr) : [], pn = b.payments.map(payStr);
  pn.filter(x => !po.includes(x)).forEach(x => out.push({ k: 'Upcoming payment', o: undefined, n: x }));
  po.filter(x => !pn.includes(x)).forEach(x => out.push({ k: 'Upcoming payment', o: x, n: undefined }));
  return out;
}
// Payment log: adds only payments not already logged. The same date + account + amount is the
// same payment, whichever list it came from (Recent or All Payments). A second identical one in
// the SAME paste counts as a genuine extra payment.
const normAcct = a => String(a || '').toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim();
const ledgerBase = e => [e.date, normAcct(e.account), String(e.amount || '').replace(/\s+/g, '')].join('|');
function fixLedger(l) {
  const cnt = {};
  return (Array.isArray(l) ? l : []).slice().sort((x, y) => (x.added || 0) - (y.added || 0)).map(e => {
    const b = ledgerBase(e); cnt[b] = (cnt[b] || 0) + 1; e.id = b + '#' + cnt[b]; return e;
  });
}
function mergeLedger(ledger, recent, ts) {
  const counts = {}, logged = [];
  for (const r of recent) {
    const base = ledgerBase(r);
    const k = counts[base] = (counts[base] || 0) + 1;
    const id = base + '#' + k;
    const ex = ledger.find(e => e.id === id);
    if (!ex) {
      ledger.push({ id, date: r.date, status: r.status, note: r.note || '', account: r.account, amount: r.amount, added: ts });
      logged.push({ kind: 'new', entry: recStr(r) });
    } else {
      if (!ex.note && r.note) ex.note = r.note;
      if (ex.status !== r.status) { logged.push({ kind: 'status', from: ex.status, entry: recStr(r) }); ex.status = r.status; }
    }
  }
  return logged;
}
function loggedChanges(s) {
  return (s.logged || []).map(l => l.kind === 'new'
    ? { k: 'Payment logged', o: undefined, n: l.entry }
    : { k: 'Payment status', o: l.from, n: l.entry });
}
const dateVal = d => { const m = String(d).match(/^(\d\d)\/(\d\d)\/(\d\d)$/); return m ? +('20' + m[3] + m[1] + m[2]) : 0; };

  return { link: '', labels: '', snaps: [], ledger: [] };
}

}

function safeLink(u) {
  u = (u || '').trim(); if (!u) return '';
  if (!/^[a-z][a-z0-9+.-]*:/i.test(u)) u = 'https://' + u;
  return /^https?:\/\//i.test(u) ? u : '';
}
function money(s) { const m = String(s || '').replace(/,/g, '').match(/-?\$?\s*(-?\d+(\.\d+)?)/); return m ? parseFloat(m[1]) : null; }
function fmtDelta(o, n) {
  const a = money(o), b = money(n);
  if (a == null || b == null || a === b || !/\$/.test(o + n)) return '';
  const d = b - a;
  return ' <span class="' + (d < 0 ? 'ok' : 'warn') + '">(' + (d > 0 ? '+' : '−') + '$' + Math.abs(d).toFixed(2) + ')</span>';
}
function daysInfo(v) {
  const n = parseInt(v, 10);
  if (isNaN(n)) return { t: v || '—', c: '' };
  if (n < 0) return { t: (-n) + ' day' + (n === -1 ? '' : 's') + ' until due', c: 'ok' };
  if (n === 0) return { t: 'Due today', c: 'warn' };
  return { t: n + ' day' + (n === 1 ? '' : 's') + ' past due', c: 'bad' };
}

// Network first (updates arrive), cache when offline.
self.addEventListener('fetch', e => {
  const req = e.request;
  if (req.method !== 'GET' || new URL(req.url).origin !== location.origin) return;
  e.respondWith(
    fetch(req)
      .then(res => {
        const copy = res.clone();
        caches.open(CACHE).then(c => c.put(req, copy));
        return res;
      })
      .catch(() => caches.match(req).then(r => r || caches.match('./index.html')))
  );
});

const $ = id => document.getElementById(id);

// Each copy of this runner lives in its own folder; the folder name is its storage slot.
const dirs = location.pathname.split('/').filter(Boolean);
if (dirs.length && /\.[a-z0-9]+$/i.test(dirs[dirs.length - 1])) dirs.pop();
const SLOT = (dirs.pop() || 'root').toLowerCase();
const K_CODE = 'runner:' + SLOT + ':code', K_STATE = 'runner:' + SLOT + ':state';

const getLS = k => { try { return localStorage.getItem(k); } catch (e) { return null; } };
function setLS(k, v) { try { localStorage.setItem(k, v); return true; } catch (e) { return false; } }
function readState() { try { const s = JSON.parse(getLS(K_STATE) || '{}'); return s && typeof s === 'object' ? s : {}; } catch (e) { return {}; } }

let curState = readState(), pending = null, timer = 0;
function toast(t) {
  const d = document.createElement('div'); d.id = 'toast'; d.textContent = t; document.body.appendChild(d);
  setTimeout(() => d.remove(), 3500);
}
function flush() {
  clearTimeout(timer);
  if (pending === null) return;
  const ok = setLS(K_STATE, JSON.stringify(pending));
  pending = null;
  if (!ok) toast('Could not save: storage is full or blocked');
}
addEventListener('pagehide', flush);
document.addEventListener('visibilitychange', () => { if (document.hidden) flush(); });

// ---- Runs inside the pasted app (before its own scripts). Gives it a saved localStorage and
// ---- hands link clicks to the runner. Must be self-contained: it is converted to text.
function shim(S) {
  function post(m) { try { m.runner = 1; parent.postMessage(m, '*'); } catch (e) {} }
  function mk(persist) {
    var d = persist ? S : {};
    var api = {
      getItem: function (k) { return Object.prototype.hasOwnProperty.call(d, k) ? d[k] : null; },
      setItem: function (k, v) { d[k] = String(v); if (persist) post({ t: 'save', state: d }); },
      removeItem: function (k) { delete d[k]; if (persist) post({ t: 'save', state: d }); },
      clear: function () { for (var k in d) delete d[k]; if (persist) post({ t: 'save', state: d }); },
      key: function (i) { return Object.keys(d)[i] || null; }
    };
    Object.defineProperty(api, 'length', { get: function () { return Object.keys(d).length; } });
    return api;
  }
  try { Object.defineProperty(window, 'localStorage', { value: mk(true), configurable: true }); } catch (e) {}
  try { Object.defineProperty(window, 'sessionStorage', { value: mk(false), configurable: true }); } catch (e) {}
  document.addEventListener('click', function (e) {
    if (e.defaultPrevented || e.button) return;
    var a = e.target && e.target.closest ? e.target.closest('a[href]') : null;
    if (!a) return;
    var h = a.getAttribute('href');
    if (!h || h.charAt(0) === '#' || /^(javascript|mailto|tel|blob|data):/i.test(h)) return;
    e.preventDefault();
    post({ t: 'open', href: h });
  }, false);
}
function inject(code, state) {
  const json = JSON.stringify(state).replace(/</g, '\\u003c');
  const tag = '<script>(' + shim.toString() + ')(' + json + ');<\/script>';
  let m;
  if ((m = code.match(/<head[^>]*>/i))) return code.slice(0, m.index + m[0].length) + tag + code.slice(m.index + m[0].length);
  if ((m = code.match(/^\s*<!doctype[^>]*>\s*(<html[^>]*>)?/i))) return code.slice(0, m[0].length) + tag + code.slice(m[0].length);
  if ((m = code.match(/<html[^>]*>/i))) return code.slice(0, m.index + m[0].length) + tag + code.slice(m.index + m[0].length);
  return tag + code;
}

const frame = $('frame');
function run(code) {
  curState = readState();
  frame.srcdoc = inject(code, curState);
  frame.hidden = false; $('menuBtn').hidden = false; $('setup').hidden = true;
  const t = code.match(/<title[^>]*>([^<]*)<\/title>/i);
  document.title = (t && t[1].trim()) || 'App Runner';
}

addEventListener('message', e => {
  if (e.source !== frame.contentWindow) return;

  if (!d || !d.runner) return;
  if (d.t === 'save' && d.state && typeof d.state === 'object') {
    curState = d.state; pending = d.state;
    clearTimeout(timer); timer = setTimeout(flush, 400);
  } else if (d.t === 'open' && typeof d.href === 'string') {
    let u; try { u = new URL(d.href, location.href); } catch (err) { return; }
    if (!/^https?:$/.test(u.protocol)) return;
    if (u.origin === location.origin) { flush(); location.href = u.href; }
    else window.open(u.href, '_blank', 'noopener');
  }
});

  if (!setLS(K_CODE, code)) { $('msg').innerHTML = '<span class="bad">Could not save: too large or storage blocked.</span>'; return; }
  run(code);
};

  else flush();
}

// ---- small menu: backup and restore

  const kb = n => (n / 1024).toFixed(1) + ' KB';
  $('info').textContent = 'Slot: ' + SLOT + ' · app ' + kb((getLS(K_CODE) || '').length) + ' · data ' + kb(JSON.stringify(curState).length);

  a.click(); setTimeout(() => URL.revokeObjectURL(a.href), 2000);
};

$('rsFile').onchange = async e => {
  const f = e.target.files[0]; if (!f) return;
  try {
    const d = JSON.parse(await f.text());
    if (!d || d.runner !== 1 || typeof d.code !== 'string' || !d.code) throw new Error('Not a runner backup');
    if (!confirm('Replace this app and its data with the backup from ' + (d.savedAt || 'unknown date') + '?')) return;
    if (!setLS(K_CODE, d.code) || !setLS(K_STATE, JSON.stringify(d.state || {}))) throw new Error('Could not save (storage full)');
    location.reload();
  } catch (err) { alert('Restore failed: ' + err.message); }
  e.target.value = '';
};

// ---- start
if (navigator.storage && navigator.storage.persist) navigator.storage.persist().catch(() => {});
if ('serviceWorker' in navigator && /^https?:$/.test(location.protocol)) {
  navigator.serviceWorker.register('sw.js', { scope: './' }).catch(() => {});
}
const saved = getLS(K_CODE);
if (saved) run(saved); else $('setup').hidden = false;

// Offline helper for the App Runner (this folder only). Uses the repo-wide shared cache name.
const CACHE = 'tiny-apps';
const FILES = ['./', './index.html', './manifest.webmanifest', './icon.png'];

self.addEventListener('install', e => {
  e.waitUntil(caches.open(CACHE).then(c => c.addAll(FILES)).then(() => self.skipWaiting()));
});

self.addEventListener('activate', e => {
  e.waitUntil(self.clients.claim());
});

// Network first (updates arrive), cache when offline.
self.addEventListener('fetch', e => {
  const req = e.request;
  if (req.method !== 'GET' || new URL(req.url).origin !== location.origin) return;
  e.respondWith(
    fetch(req)
      .then(res => {
        const copy = res.clone();
        caches.open(CACHE).then(c => c.put(req, copy));
        return res;
      })
      .catch(() => caches.match(req).then(r => r || caches.match('./index.html')))
  );
});

# Tiny Apps repo: instructions for Copilot

This repo is a hub plus a collection of tiny offline web apps (PWAs), hosted on GitHub Pages.
No build step, no frameworks, no npm. Plain HTML, CSS and JavaScript only.

## Layout

```
index.html              HUB. The landing page, and itself an installable PWA.
sw.js                   Hub service worker.
manifest.webmanifest    Hub manifest ("Tiny Apps").
icon.png                Hub icon.
.github/copilot-instructions.md   This file.
<app>/                  One folder per tiny app (example: loan/)
  index.html            The whole app in ONE file (HTML + CSS + JS inline).
  sw.js                 This app's own service worker.
  manifest.webmanifest  Name, colors and icon, so it can also be installed on its own.
  icon.png              App icon.
```

## How the hub works

- The hub lists every app from the `APPS` array at the top of the script in the root `index.html`.
- The hub can be added to the home screen. Its scope is the whole site, so tapping an app opens it inside the hub.
- Every app can also be installed on its own from its own folder.
- Stay up to date: on load (and with the "Check for updates" button) the hub re-downloads
  its own files and every app's `files` list (bypassing the HTTP cache) into the shared cache,
  so all apps work offline and stay current without opening each one.
- Every service worker in this repo uses the SAME cache name: `tiny-apps`. Do not change it,
  and do not create per-app cache names. Workers are network first, falling back to the cache offline.

## The runner (`runner/`)

A template app. The first launch asks the user to paste HTML or choose an HTML/text file; after that
this copy loads that app every time with no prompt, so it simply becomes that app.
- The pasted code is stored in localStorage under `runner:<folder name>:code`, and the app's own data
  under `runner:<folder name>:state`. The folder name is the slot, so every copy needs its own folder.
- The app runs in a sandboxed iframe (`srcdoc`). The runner injects a small shim (`shim()`) that gives
  the app a `localStorage` that is saved by the runner (a hidden scratch pad), and sends link clicks to the runner.
  Pasted apps need no code changes, as long as they use localStorage (not IndexedDB or their own service worker).
- A small ⋯ button (bottom left) opens Save backup / Restore backup / Back to Tiny Apps.
- To make another app: copy the whole `runner/` folder to a new folder name (for example `budget/`), then add
  a hub entry for it in the root `index.html` `APPS` list. Do not edit the runner code per copy.
- Do not add "reset" or "replace app" features unless asked.

## Rules

- Every app lives in its own folder. Never mix one app's files into another's folder or the root.
- Each app is self-contained: one `index.html` with inline CSS and JS. Use relative paths only.
- Each app registers its own `sw.js` with `scope: './'` from its own folder.
- Data is stored on the device (localStorage or IndexedDB). No servers, no accounts, no tracking.
- Use a unique storage key per app (example: `loanTracker.v1`) because all apps share one origin.
- Every app has a "← Back to Tiny Apps" button at the very bottom (a link to `../`) that asks the user
  to confirm with a pop-up (`confirm()`) before leaving, and warns if there is unsaved input. Copy it from `loan/index.html` (`#homeLink`).
- Mobile first. Simple, readable, light and dark mode. Keep files small.
- Icons: PNG, 512x512, no more than three colors (brand blue `#0025b8`, green `#098a11`, white), as small as possible.

## Adding a new app

1. Create `<app>/` with `index.html`, `sw.js`, `manifest.webmanifest` and `icon.png`.
   Copy `loan/sw.js` and `loan/manifest.webmanifest` and change the name, `start_url` stays `./`.
2. Add one entry to `APPS` in the root `index.html`:
   `{ name: 'Name', file: '<app>/', desc: 'One line', files: ['<app>/', '<app>/index.html', '<app>/manifest.webmanifest', '<app>/icon.png'] }`
   The `files` list must name every file the app needs offline. The hub tile always shows that app folder's own `icon.png`, so there is no icon setting.
3. Do not edit other apps while adding a new one.

## Current apps

- `loan/` Loan Tracker. Paste loan account text. It saves snapshots, shows what changed, keeps a
  de-duplicated payment log (the same date, account and amount is never logged twice), and has a
  link to the lender site. It understands four pasted screens: the main account page (with Upcoming and
  Recent payments), Additional Payment Details, Recurring Autopay Scheduled, and All Payments. The parser
  is the `parseText` function between the `/*PARSER*/` markers. A "View all payments" button shows the
  full log grouped by month.
  Each paste is saved as a timestamped snapshot, and the current status is all snapshots layered
  oldest to newest (`mergedState`): newer values win on conflicts, and anything a newer paste leaves
  out (or shows as `-`) is kept from earlier pastes. Never let a partial paste erase data.
  The "days until/past due" count is calculated live (`daysNow`) from the most recently pasted of
  Days Past Due (moved forward by the days since that paste) or Current Due Date, so it changes daily.
